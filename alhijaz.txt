DROP DATABASE IF EXISTS alhijaz;
CREATE DATABASE alhijaz;
USE alhijaz;

-- Production related tables
CREATE TABLE production_line (
    line_num VARCHAR(50) PRIMARY KEY,
    mechanism VARCHAR(100)
);

CREATE TABLE brand (
    brand_id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100),
    roll_color VARCHAR(50)
);

-- Product related tables
CREATE TABLE chocolate (
    chocolate_id INT PRIMARY KEY AUTO_INCREMENT,
    type VARCHAR(50),
    shape VARCHAR(50),
    measure_unit VARCHAR(20),
    unit_price DECIMAL(10,2),
    reorder_level INT,
    barcode VARCHAR(50),
    created_date DATE,
    expiry_date DATE,
    number_of_units_in_stock INT,
    brand_id INT,
    image_path VARCHAR(255),
    FOREIGN KEY (brand_id) REFERENCES brand(brand_id) ON DELETE SET NULL
);

-- Inventory related tables
CREATE TABLE stock (
    stock_id INT PRIMARY KEY AUTO_INCREMENT,
    type VARCHAR(50),
    floor_no VARCHAR(20),
    max_capacity INT
);

CREATE TABLE ingredient (
    ing_id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100),
    reorder_level INT,
    unit_price DECIMAL(10,2),
    number_of_units_in_stock INT
);

-- Personnel related tables
CREATE TABLE department (
    dept_num VARCHAR(50) PRIMARY KEY,
    name VARCHAR(100)
);

CREATE TABLE employee (
    employee_id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100),
    gender VARCHAR(20),
    age INT,
    role VARCHAR(50),
    shift_type VARCHAR(50),
    working_hours INT,
    salary DECIMAL(10,2),
    dept_num VARCHAR(50),
    FOREIGN KEY (dept_num) REFERENCES department(dept_num) ON DELETE SET NULL
);

CREATE TABLE sales_rep (
    sales_rep_id INT PRIMARY KEY,
    region VARCHAR(100),
    FOREIGN KEY (sales_rep_id) REFERENCES employee(employee_id) ON DELETE CASCADE
);

-- Customer related tables
CREATE TABLE client (
    client_id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100),
    location VARCHAR(200),
    phone_number VARCHAR(20),
    email VARCHAR(100)
);

-- Order related tables
CREATE TABLE ordert (
    order_id INT PRIMARY KEY AUTO_INCREMENT,
    order_date DATE,
    arrival_date DATE,
    client_id INT,
    sales_rep_id INT,
    FOREIGN KEY (client_id) REFERENCES client(client_id) ON DELETE SET NULL,
    FOREIGN KEY (sales_rep_id) REFERENCES sales_rep(sales_rep_id) ON DELETE SET NULL
);

CREATE TABLE orderdetail (
    order_id INT,
    chocolate_id INT,
    payment_method VARCHAR(50),
    quantity INT,
    unit_price DECIMAL(10,2),
    discount DECIMAL(5,2),
    PRIMARY KEY(order_id, chocolate_id),
    FOREIGN KEY (order_id) REFERENCES ordert(order_id) ON DELETE CASCADE,
    FOREIGN KEY (chocolate_id) REFERENCES chocolate(chocolate_id) ON DELETE CASCADE
);

-- Relationship tables
CREATE TABLE stock_chocolate (
    stock_id INT,
    chocolate_id INT,
    quantity INT,
    PRIMARY KEY (stock_id, chocolate_id),
    FOREIGN KEY (stock_id) REFERENCES stock(stock_id) ON DELETE CASCADE,
    FOREIGN KEY (chocolate_id) REFERENCES chocolate(chocolate_id) ON DELETE CASCADE
);

CREATE TABLE chocolate_ingredient (
    chocolate_id INT,
    material_id INT,
    quantity DECIMAL(10,2),
    PRIMARY KEY (chocolate_id, material_id),
    FOREIGN KEY (chocolate_id) REFERENCES chocolate(chocolate_id) ON DELETE CASCADE,
    FOREIGN KEY (material_id) REFERENCES ingredient(ing_id) ON DELETE CASCADE
);

CREATE TABLE production_brand (
    line_num VARCHAR(50),
    brand_id INT,
    PRIMARY KEY (line_num, brand_id),
    FOREIGN KEY (line_num) REFERENCES production_line(line_num) ON DELETE CASCADE,
    FOREIGN KEY (brand_id) REFERENCES brand(brand_id) ON DELETE CASCADE
);
DROP TABLE IF EXISTS cart;
CREATE TABLE cart (
    cart_id INT AUTO_INCREMENT PRIMARY KEY,
    client_id INT NOT NULL,
    product_id INT NOT NULL,
    quantity INT DEFAULT 1,
    FOREIGN KEY (client_id) REFERENCES client(client_id),  -- Changed from clients to client
    FOREIGN KEY (product_id) REFERENCES chocolate(chocolate_id)
);
-- Add stock_ingredient table which exists in the second schema
CREATE TABLE stock_ingredient (
    stock_id INT,
    ing_id INT,
    quantity INT,
    PRIMARY KEY (stock_id, ing_id),
    FOREIGN KEY (stock_id) REFERENCES stock(stock_id) ON DELETE CASCADE,
    FOREIGN KEY (ing_id) REFERENCES ingredient(ing_id) ON DELETE CASCADE
);

-- Add security tables from second schema
CREATE TABLE emp_security (
    emp_id INT PRIMARY KEY,
    password CHAR(8),
    FOREIGN KEY (emp_id) REFERENCES employee(employee_id)
);

CREATE TABLE manager_security (
    mng_id INT PRIMARY KEY,
    password CHAR(8)
);

CREATE TABLE client_security (

-- Create stored procedure for client login
DELIMITER //
CREATE PROCEDURE client_login(
    IN p_client_id INT,
    IN p_password VARCHAR(255)
)
BEGIN
    DECLARE client_exists INT;
    
    SELECT COUNT(*) INTO client_exists
    FROM client
    WHERE client_id = p_client_id AND password = p_password;
    
    IF client_exists > 0 THEN
        -- Return client information
        SELECT 
            c.client_id,
            c.name,
            c.location,
            c.phone_number,
            c.email,
            o.order_id,
            o.order_date,
            o.arrival_date,
            ch.type as chocolate_type,
            od.quantity,
            od.unit_price,
            od.discount
        FROM client c
        LEFT JOIN ordert o ON c.client_id = o.client_id
        LEFT JOIN orderdetail od ON o.order_id = od.order_id
        LEFT JOIN chocolate ch ON od.chocolate_id = ch.chocolate_id
        WHERE c.client_id = p_client_id
        ORDER BY o.order_date DESC;
    ELSE
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Invalid client ID or password';
    END IF;
END //
DELIMITER ;
